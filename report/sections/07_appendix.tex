\subsection{Supplementary Algorithms and Listings}

\begin{algorithm}[H]
\caption{Drug--Disease Success Labeling}\label{alg:labeling}
\begin{algorithmic}% [1] for line numbers
\REQUIRE Clinical trial data for a drug--disease indication
\ENSURE Success, fail or unknown label

\IF{phase 4 is passed}
    \STATE label $\gets$ \texttt{success}
\ELSIF{any clinical trial exists and has results}
    \STATE Compute median p-value of trials
    \IF{median p-value $< 0.05$}
        \STATE trial label $\gets$ \texttt{success}
    \ELSIF{median p-value $> 0.2$}
        \STATE trial label $\gets$ \texttt{fail}
    \ELSE
        \STATE trial label $\gets$ \texttt{unknown}
    \ENDIF

    \STATE Let $P_{\max}$ be the highest phase tried
    \STATE Let $S$ and $F$ be the number of successes and fails in phase $P_{\max}$
    \IF{$P_{\max} \ge 3$ \textbf{and} $S \ge 2 \times F$}
        \STATE label $\gets$ \texttt{success}
    \ELSIF{$F \ge 2 \times S$}
        \STATE label $\gets$ \texttt{fail}
    \ELSE
        \STATE label $\gets$ \texttt{unknown}
    \ENDIF
\ENDIF
\IF{label is still unknown}
    \STATE Let $D_{\text{latest}}$ be the latest trial date
    \IF{$D_{\text{latest}} < 2020$}
        \STATE label $\gets$ \texttt{fail}
    \ELSE
        \STATE label $\gets$ \texttt{unknown}
    \ENDIF
\ENDIF
\RETURN label
\end{algorithmic}
\end{algorithm}
